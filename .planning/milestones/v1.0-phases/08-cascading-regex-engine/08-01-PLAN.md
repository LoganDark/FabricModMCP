    ```typescript
    export interface CascadeStep {
    	step: number;        // 1-based step number
    	pattern: string;     // Original pattern string (including any (?flags) prefix)
    	status: 'success' | 'failed';
    	matched?: string;    // The matched text (only on success)
    	offset?: number;     // Absolute character offset in source (only on success)
    	length?: number;     // Length of matched text (only on success)
    }

    export interface CascadeSuccess {
    	success: true;
    	steps: CascadeStep[];
    	offset: number;      // Final match absolute offset
    	line: number;        // 1-based line number
    	column: number;      // 1-based column number
    }

    export interface CascadeFailure {
    	success: false;
    	steps: CascadeStep[];
    	failedStep: number;  // 1-based step number that failed
    	error?: string;      // Optional error message (for invalid regex syntax)
    }

    export type CascadeResult = CascadeSuccess | CascadeFailure;
    ```

    **Main function signature:**
    ```typescript
    export function cascadeRegex(source: string, patterns: string[]): CascadeResult
    ```

    **Implementation details:**

    1. **Empty patterns check:** If `patterns.length === 0`, return a `CascadeFailure` with `failedStep: 0`, empty steps, and `error: "No patterns provided"`.

    2. **Flag prefix parsing:** Before constructing each RegExp, check if the pattern starts with `(?flags)` where flags are one or more of `i`, `m`, `s`. Parse with: `const prefixMatch = pattern.match(/^\(\?([imsu]+)\)/);`. If matched, strip the prefix from the pattern body and pass the flags to `new RegExp(body, flags)`. If no prefix, use `new RegExp(pattern)` with no flags.

    3. **Regex construction error handling:** Wrap `new RegExp()` in try/catch. On SyntaxError, return `CascadeFailure` with `failedStep` set to the current 1-based step number, steps containing all prior successes, and `error` set to the SyntaxError message string.

    4. **Cascade execution loop:**
       - Track `currentText` (starts as full `source`) and `baseOffset` (starts at 0).
       - For each pattern: compile regex, call `regex.exec(currentText)`.
       - On match: record a success step with `offset: baseOffset + match.index`, `length: match[0].length`, `matched: match[0]`. Update `currentText = match[0]` and `baseOffset += match.index`.
       - On no match: record a failed step (no matched/offset/length fields), return `CascadeFailure` with `failedStep` = current step number.

    5. **Line/column computation:** After all steps succeed, compute 1-based line and column from the final absolute offset:
       ```typescript
       function offsetToLineColumn(source: string, offset: number): { line: number; column: number } {
       	let line = 1;
       	let lastNewline = -1;
       	for (let i = 0; i < offset; i++) {
       		if (source[i] === '\n') {
       			line++;
       			lastNewline = i;
       		}
       	}
       	return { line, column: offset - lastNewline };
       }
       ```

    6. **Return CascadeSuccess** with the final step's offset/line/column and the full steps trace array.
