export interface PrimitiveType { kind: "primitive"; name: string; }

  name: string | null;
  type: TypeReference;

  kind: "method";
  parameters: ParameterInfo[];
  returnType: TypeReference | null;  // null for constructors
}

export interface FieldReference {
  kind: "field";
  fieldType: TypeReference;
}

export type MemberReference = MethodReference | FieldReference;

    ```typescript
    export async function parseDetail(
      detail: string | null,
      symbolKind: string,
      resolveType: (simpleName: string) => Promise<TypeReference>,
    ): Promise<MemberReference | null>
    ```

    Implementation steps:
    1. If `detail` is null or empty string -> return null
    2. Determine if field or method based on `symbolKind`:
       - "method", "constructor" -> parse as method
       - "field", "constant", "enumMember" -> parse as field
       - Other -> return null
    3. **Field parsing:** Call `resolveTypeToken(detail, resolveType)` to get TypeReference, return `{ kind: "field", fieldType }`
    4. **Method parsing:**
       - If detail does not start with `(` -> no-arg method, treat entire string as return type after resolving
         - Actually: no-arg method detail is `"void"` or similar. Check if it contains `(` -- if no parens, it's a no-arg method with the detail as return type (e.g., `"void"`)
       - Split on ` : ` to separate params section from return type
       - If no ` : ` found -> constructor (no return type), entire string is params in parens
       - Parse params: strip outer `()`, split by comma at depth 0 (track `<>` nesting depth), trim each
       - For each param: call `resolveTypeToken(param, resolveType, isLastParam)` -- isLastParam enables vararg detection
       - Return type: if symbolKind is "constructor" -> null, else resolve the return type string
    5. **resolveTypeToken helper function** (not exported):
       - Strip leading annotations: regex `/^(?:@\w+(?:\([^)]*\))?\s+)*/` (repeat for multiple annotations)
       - Strip generic type arguments: depth-counting loop from first `<` to balanced `>`
       - Check for trailing `[]` -> wrap result in ArrayType (handle multi-dimensional: `int[][]`)
       - Check for trailing `...` -> wrap result in VarargType (only valid for last param)
       - Call `resolveType(simpleName)` with the cleaned name
    6. **splitParams helper function** (not exported):
       - Track angle bracket depth, only split on commas at depth 0
       - Return trimmed param strings
