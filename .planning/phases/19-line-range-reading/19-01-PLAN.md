function extractContext(
	source: string,
	line: number,
	linesBefore: number,
	linesAfter: number,
): LocateResultContext {
	const lines = source.split('\n');
	const totalLines = lines.length;
	const startLine = Math.max(1, line - linesBefore);
	const endLine = Math.min(totalLines, line + linesAfter);
	const text = lines.slice(startLine - 1, endLine).join('\n');
	return { text, startLine, endLine };
}
```

export interface LineSliceResult {
	source: string;
	startLine: number;
	endLine: number;
	totalLineCount: number;
	truncated: boolean;
}

export function sliceLines(
	sourceText: string,
	requestedStartLine?: number,
	requestedLineCount?: number,
): LineSliceResult
```

Implementation rules:
- Use `sourceText.split('\n')` for line splitting (project convention, do NOT use regex like `/\r?\n/`)
- `totalLineCount` = `lines.length` (always, even for empty string which gives `[""]` = 1)
- When neither param provided: return full source unchanged, startLine=1, endLine=totalLineCount, truncated=false
- When startLine provided and > totalLineCount: return empty string, startLine=requestedStartLine, endLine=requestedStartLine-1, truncated=true
- Compute effective start as `requestedStartLine ?? 1`
- Compute effective end as: if lineCount given, `Math.min(start + lineCount - 1, totalLineCount)`, else `totalLineCount`
- Extract with `lines.slice(start - 1, end).join('\n')` (1-based to 0-based: subtract 1 from start, end is exclusive so use end directly since it's inclusive 1-based)
- `truncated` = true if `start !== 1 || end !== totalLineCount` (any narrowing from full file)

Then create `tests/browsing/line-slicer.test.ts` with all behaviors listed above. Use vitest `describe`/`it`/`expect` pattern. Tab indentation.
