2. In `src/project/dependency-discovery.ts`, modify `addDependencyEntry` signature to accept a new `chain: string[] = []` parameter:
   ```typescript
   async function addDependencyEntry(
   	deps: Map<string, DependencyEntry>,
   	group: string, artifact: string, version: string,
   	category: DependencyEntry['category'],
   	chain: string[] = [],
   ): Promise<void> {
   	const id = `${group}:${artifact}`;
   	const existing = deps.get(id);
   	if (existing) {
   		if (chain.length > 0) {
   			existing.provenanceChains.push(chain);
   		}
   		return;
   	}
   	const sourcesJarPath = await findSourcesJar(group, artifact, version);
   	deps.set(id, {
   		id, group, artifact, version, category, sourcesJarPath,
   		available: sourcesJarPath !== null,
   		provenanceChains: chain.length > 0 ? [chain] : [],
   	});
   }
   ```

3. Modify `followTransitiveDeps` signature to accept `chain: string[]` parameter:
   ```typescript
   async function followTransitiveDeps(
   	deps: Map<string, DependencyEntry>,
   	group: string, artifact: string, version: string,
   	visited: Set<string>, depth: number,
   	chain: string[],
   ): Promise<void>
   ```
   Inside the POM deps loop, compute `const depId = \`\${dep.groupId}:\${dep.artifactId}\`;` and `const newChain = [...chain, depId];`, then pass `newChain` to both `addDependencyEntry` and the recursive `followTransitiveDeps` call.
