**src/state/project-store.ts:**

```typescript
import type { LoadedProject } from '../project/types.js';

export class ProjectStore {
	private projects = new Map<string, LoadedProject>();

	get(name: string): LoadedProject | undefined {
		return this.projects.get(name);
	}

	set(name: string, project: LoadedProject): void {
		this.projects.set(name, project);
	}

	has(name: string): boolean {
		return this.projects.has(name);
	}

	list(): LoadedProject[] {
		return Array.from(this.projects.values());
	}

	delete(name: string): boolean {
		return this.projects.delete(name);
	}

	get size(): number {
		return this.projects.size;
	}
}

export const projectStore = new ProjectStore();
```

**src/project/loader.ts** -- exports `loadProject(projectPath: string): Promise<LoadedProject>`:

1. `const absolutePath = path.resolve(projectPath)` -- resolves `.` and relative paths
2. `const name = path.basename(absolutePath)` -- extract directory name
3. Validate directory exists: `await fs.stat(absolutePath)` -- if ENOENT, throw `DomainError('PROJECT_NOT_FOUND', 'Project directory does not exist', [absolutePath], ['Check the path and try again'])`
4. Validate it's a directory: check `stat.isDirectory()` -- if not, throw `DomainError('PROJECT_NOT_DIRECTORY', 'Path is not a directory', [absolutePath])`
5. Read `gradle.properties`: `await fs.readFile(path.join(absolutePath, 'gradle.properties'), 'utf-8')` -- if ENOENT, throw `DomainError('GRADLE_PROPERTIES_NOT_FOUND', 'gradle.properties not found -- is this a Gradle project?', [path.join(absolutePath, 'gradle.properties')], ['Ensure this is a Fabric/Loom project root directory'])`
6. Parse properties: `const properties = parseGradleProperties(propertiesContent)`
7. Read `build.gradle.kts`: `await fs.readFile(path.join(absolutePath, 'build.gradle.kts'), 'utf-8')` -- if ENOENT, throw `DomainError('BUILD_GRADLE_NOT_FOUND', 'build.gradle.kts not found', [path.join(absolutePath, 'build.gradle.kts')], ['This server only supports Kotlin DSL (build.gradle.kts), not Groovy (build.gradle)'])`
8. Parse gradle: `const gradleConfig = parseBuildGradle(buildGradleContent, properties)`
9. Resolve sources jar path: `const sourcesJarPath = resolveSourcesJarPath(gradleConfig)`
10. Check sources jar exists: `const sourcesJarExists = await fileExists(sourcesJarPath)` (helper: try `fs.access`, return boolean)
11. If sources jar does NOT exist, throw `DomainError('SOURCES_JAR_NOT_FOUND', "Couldn't find Minecraft sources jar -- have you run genSources?", [sourcesJarPath], ['Run ./gradlew genSources in your project directory', 'Check that your Loom cache is at ~/.gradle/caches/fabric-loom/'])`
12. Read fabric.mod.json: `await fs.readFile(path.join(absolutePath, 'src', 'main', 'resources', 'fabric.mod.json'), 'utf-8')` -- if ENOENT, throw `DomainError('FABRIC_MOD_NOT_FOUND', 'fabric.mod.json not found', [path.join(absolutePath, 'src', 'main', 'resources', 'fabric.mod.json')], ['Ensure this is a Fabric mod project with src/main/resources/fabric.mod.json'])`
13. Parse fabric mod: `const fabricMod = parseFabricMod(fabricModContent)`
14. Return `{ name, rootPath: absolutePath, gradleConfig, sourcesJar: { path: sourcesJarPath, exists: true }, fabricMod, dependencyJars: new Map() }` -- dependencyJars is an empty map for now (Phase 3 will populate it)

```typescript
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import { createServer } from './server.js';
import { registerAllTools } from './tools/index.js';
import { parseCli } from './cli/args.js';
import { logger } from './logging/logger.js';
import { loadProject } from './project/loader.js';
import { projectStore } from './state/project-store.js';

const args = parseCli(process.argv.slice(2));
logger.setLevel(args.logLevel);

// Require --project flag (Phase 4 will allow empty start)
if (!args.project) {
	logger.error('Missing required --project flag. Usage: minecraft-dev-mcp --project /path/to/mod');
	process.exit(1);
}

// Load project before connecting transport
try {
	const project = await loadProject(args.project);
	projectStore.set(project.name, project);
	logger.info('Project loaded', {
		name: project.name,
		minecraftVersion: project.gradleConfig.minecraftVersion,
		mappingEra: project.gradleConfig.mappingEra,
		sourcesJar: project.sourcesJar.path,
	});
} catch (error) {
	if (error instanceof Error) {
		logger.error(`Failed to load project: ${error.message}`);
		if ('tried' in error && Array.isArray((error as any).tried)) {
			logger.error('Paths tried:', (error as any).tried);
		}
		if ('suggestions' in error && Array.isArray((error as any).suggestions)) {
			for (const suggestion of (error as any).suggestions as string[]) {
				logger.error(`  Suggestion: ${suggestion}`);
			}
		}
	}
	process.exit(1);
}

const server = createServer();
registerAllTools(server);

const transport = new StdioServerTransport();
await server.connect(transport);

logger.info('Server started', { transport: 'stdio' });

process.on('SIGINT', async () => {
	logger.info('Shutting down...');
	await server.close();
	process.exit(0);
});
```

Key behaviors:
- If `--project` is not provided, logs a clear error to stderr and exits with code 1
- If project loading fails (any DomainError), logs the error message, tried paths, and suggestions to stderr, then exits with code 1
- On success, logs project name, MC version, mapping era, and sources jar path to stderr (info level)
- Project is loaded and stored BEFORE the MCP server is created and transport connected
- Import `DomainError` is not needed -- the catch block uses duck typing on `tried` and `suggestions` properties
